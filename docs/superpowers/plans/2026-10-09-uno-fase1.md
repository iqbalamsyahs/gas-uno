# UNO Fase 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Memindahkan prototipe UNO (`index.html`, 1 pemain lawan 3 bot) ke Next.js dengan tampilan meja kartu klasik, bisa dimainkan di web (Vercel) lalu di dalam Discord dengan login.

**Architecture:** Logika game ada di `lib/uno.ts` sebagai modul TypeScript murni (state tidak diubah di tempat, aksi lewat `apply`). UI React (`'use client'`, CSS Modules + variabel CSS) hanya menerima `viewFor(state, 0)` dan mengirim aksi lewat `dispatch`. `DiscordGate` menangani SDK dan login; di luar Discord memakai mock user.

**Tech Stack:** Next.js (App Router, TypeScript), CSS Modules, `@discord/embedded-app-sdk`, Vitest, Vercel, `cloudflared`.

**Spec:** [docs/superpowers/specs/2026-10-09-uno-fase1-ui-design.md](../specs/2026-10-09-uno-fase1-ui-design.md) — baca bersama [PRD.md](../../../PRD.md).

**Penyimpangan yang disengaja dari PRD:** nickname server butuh scope `guilds.members.read`, sedangkan PRD menetapkan scope `identify` saja. Rencana ini memakai `global_name`, lalu `username` sebagai cadangan. Nickname bisa ditambahkan nanti kalau scope tambahan disetujui.

## Global Constraints

- JavaScript halaman pertama < 200 KB setelah gzip.
- Berjalan di Discord desktop, web, Android, dan iOS dengan lebar layar minimal 360 px.
- Font hanya `system-ui`. Tidak ada aset, font, atau script dari CDN luar (CSP Discord); semua di-bundle.
- Tidak ada teks fungsional di bawah 11 px. Kontras teks di felt ≥ 4.5:1. Angka di kartu kuning selalu `--ink-dark` (#1a1a1a).
- Warna selalu disertai simbol bentuk dan teks: Merah ▲ · Kuning ● · Hijau ■ · Biru ◆.
- Styling hanya CSS Modules + variabel CSS di `:root`; tidak ada library UI atau animasi.
- `prefers-reduced-motion` mematikan semua gerakan.
- Bayangan netral, tanpa glow berwarna; penanda giliran berupa garis tepi `--accent` solid.
- Jeda langkah bot 900 ms.
- `DISCORD_CLIENT_SECRET` hanya di environment variable server, tidak pernah dikirim ke browser.
- Semua teks UI berbahasa Indonesia.

```tsx
'use client';

import { useEffect, useState } from 'react';
import type { DiscordSDK } from '@discord/embedded-app-sdk';
import { MOCK_USER, type User } from '@/lib/discord-user';
import s from './DiscordGate.module.css';

type Status = { kind: 'loading' } | { kind: 'ready'; user: User } | { kind: 'error' };

let sdk: DiscordSDK | null = null;
let connecting: Promise<User> | null = null;

async function connect(): Promise<User> {
  // import dinamis supaya SDK tidak masuk bundle halaman pertama
  const { DiscordSDK } = await import('@discord/embedded-app-sdk');
  sdk ??= new DiscordSDK(process.env.NEXT_PUBLIC_DISCORD_CLIENT_ID!);
  await sdk.ready();
  return MOCK_USER;
}

// Strict Mode menjalankan effect dua kali; satu promise bersama mencegah handshake ganda
function connectOnce(): Promise<User> {
  return (connecting ??= connect().catch((e) => {
    connecting = null;
    throw e;
  }));
}

export default function DiscordGate() {
  const [status, setStatus] = useState<Status>({ kind: 'loading' });
  const [attempt, setAttempt] = useState(0);

  useEffect(() => {
    // Di browser biasa tidak ada frame_id: pakai mock user
    const inDiscord = new URLSearchParams(location.search).has('frame_id');
    const source = inDiscord ? connectOnce() : Promise.resolve(MOCK_USER);
    let cancelled = false;
    source.then(
      (user) => !cancelled && setStatus({ kind: 'ready', user }),
      (err) => {
        console.error(err);
        if (!cancelled) setStatus({ kind: 'error' });
      },
    );
    return () => {
      cancelled = true;
    };
  }, [attempt]);

  if (status.kind === 'ready') return <p className={s.screen}>Halo, {status.user.name}</p>;

  if (status.kind === 'error') {
    return (
      <div className={s.screen}>
        <p>Gagal menyambung ke Discord</p>
        <button
          type="button"
          className={s.retry}
          onClick={() => {
            setStatus({ kind: 'loading' });
            setAttempt((a) => a + 1);
          }}
        >
          Coba lagi
        </button>
      </div>
    );
  }

  return (
    <div className={s.screen} role="status">
      <p>Menyambung ke Discord…</p>
    </div>
  );
}
```

```ts
export const COLORS = ['red', 'yellow', 'green', 'blue'] as const;
export type Color = (typeof COLORS)[number];
export type Card = { c: Color | 'wild'; v: string }; // v: '0'–'9' | 'skip' | 'rev' | '+2' | 'wild' | '+4'

export type GameState = {
  names: string[];
  deck: Card[];
  discard: Card[];
  hands: Card[][];
  turn: number;
  dir: 1 | -1;
  color: Color;
  drawn: boolean; // pemain giliran ini sudah ambil kartu yang bisa dimainkan
  winner: number | null;
  log: string;
  seed: number;
};

export const COLOR_NAME: Record<Color, string> = { red: 'Merah', yellow: 'Kuning', green: 'Hijau', blue: 'Biru' };
export const SYMBOL: Record<Color, string> = { red: '▲', yellow: '●', green: '■', blue: '◆' };
const VALUE_NAME: Record<string, string> = { skip: 'Skip', rev: 'Reverse', wild: 'Wild', '+4': 'Wild +4' };

export function cardLabel(card: Card): string {
  const v = VALUE_NAME[card.v] ?? card.v;
  return card.c === 'wild' ? v : `${COLOR_NAME[card.c]} ${v}`;
}

// mulberry32: RNG ber-seed yang disimpan di state, supaya game bisa diulang di test dan simulasi
function random(s: GameState): number {
  s.seed = (s.seed + 0x6d2b79f5) >>> 0;
  let t = s.seed;
  t = Math.imul(t ^ (t >>> 15), t | 1);
  t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
}

function shuffle(s: GameState, a: Card[]): Card[] {
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(random(s) * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

export const top = (s: GameState): Card => s.discard[s.discard.length - 1];

export const matches = (card: Card, topCard: Card, color: Color): boolean =>
  card.c === 'wild' || card.c === color || card.v === topCard.v;

export const canPlay = (s: GameState, card: Card): boolean => matches(card, top(s), s.color);

function drawCards(s: GameState, seat: number, n: number) {
  for (let i = 0; i < n; i++) {
    if (!s.deck.length) s.deck = shuffle(s, s.discard.splice(0, s.discard.length - 1));
    const card = s.deck.pop();
    if (card) s.hands[seat].push(card);
  }
}

export function newGame(names: string[], seed = Date.now()): GameState {
  const deck: Card[] = [];
  for (const c of COLORS) {
    deck.push({ c, v: '0' });
    for (const v of ['1', '2', '3', '4', '5', '6', '7', '8', '9', 'skip', 'rev', '+2']) deck.push({ c, v }, { c, v });
  }
  for (let i = 0; i < 4; i++) deck.push({ c: 'wild', v: 'wild' }, { c: 'wild', v: '+4' });

  const s: GameState = {
    names, deck, discard: [], hands: names.map(() => []),
    turn: 0, dir: 1, color: 'red', drawn: false, winner: null, log: '', seed: seed >>> 0,
  };
  shuffle(s, s.deck);
  for (let p = 0; p < names.length; p++) drawCards(s, p, 7);
  // kartu pembuka harus kartu angka
  const i = s.deck.findIndex((k) => /^\d$/.test(k.v));
  s.discard = s.deck.splice(i, 1);
  s.color = top(s).c as Color;
  s.log = `Giliran ${names[0]}`;
  return s;
}
```

```ts
export type Action =
  | { type: 'play'; seat: number; index: number; color?: Color }
  | { type: 'draw'; seat: number }
  | { type: 'pass'; seat: number };

const next = (s: GameState, k = 1) => {
  const n = s.hands.length;
  return (((s.turn + s.dir * k) % n) + n) % n;
};

function endTurn(s: GameState, step: number) {
  s.turn = next(s, step);
  s.drawn = false;
}

// Aksi tidak sah mengembalikan state yang sama (===). Aturan ini juga dipakai server di Fase 2.
export function apply(state: GameState, a: Action): GameState {
  if (state.winner !== null || a.seat !== state.turn) return state;
  const s = structuredClone(state);
  const hand = s.hands[a.seat];
  const name = s.names[a.seat];

  if (a.type === 'draw') {
    if (s.drawn) return state;
    const before = hand.length;
    drawCards(s, a.seat, 1);
    if (hand.length > before && canPlay(s, hand[hand.length - 1])) {
      s.drawn = true;
      s.log = `${name} ambil kartu`;
      return s;
    }
    s.log = `${name} ambil kartu & lewat`;
    endTurn(s, 1);
    return s;
  }

  if (a.type === 'pass') {
    if (!s.drawn) return state;
    s.log = `${name} lewat`;
    endTurn(s, 1);
    return s;
  }

  const card = hand[a.index];
  if (!card || !canPlay(s, card)) return state;
  if (s.drawn && a.index !== hand.length - 1) return state; // setelah ambil, hanya kartu yang baru diambil
  if (card.c === 'wild' && !(a.color && COLORS.includes(a.color))) return state;

  hand.splice(a.index, 1);
  s.discard.push(card);
  s.color = card.c === 'wild' ? a.color! : card.c;
  s.log = `${name} main ${cardLabel(card)}${card.c === 'wild' ? ` → ${COLOR_NAME[s.color]}` : ''}`;

  if (!hand.length) {
    s.winner = a.seat;
    s.log = `${name} menang!`;
    return s;
  }
  if (hand.length === 1) s.log += ' — UNO!';

  let step = 1;
  if (card.v === 'skip') step = 2;
  if (card.v === 'rev') s.dir = s.dir === 1 ? -1 : 1;
  if (card.v === '+2' || card.v === '+4') {
    drawCards(s, next(s), card.v === '+2' ? 2 : 4);
    step = 2;
  }
  endTurn(s, step);
  return s;
}
```

```ts
export function botAction(s: GameState): Action {
  const seat = s.turn;
  const hand = s.hands[seat];
  if (s.drawn) return { type: 'play', seat, index: hand.length - 1, color: bestColor(hand) };
  // dahulukan kartu berwarna, simpan Wild untuk belakangan
  let index = hand.findIndex((k) => k.c !== 'wild' && canPlay(s, k));
  if (index < 0) index = hand.findIndex((k) => canPlay(s, k));
  return index >= 0 ? { type: 'play', seat, index, color: bestColor(hand) } : { type: 'draw', seat };
}

function bestColor(hand: Card[]): Color {
  const count = (c: Color) => hand.filter((k) => k.c === c).length;
  return COLORS.reduce((best, c) => (count(c) > count(best) ? c : best));
}

export type PlayerView = {
  seat: number;
  names: string[];
  hand: Card[];
  counts: number[];
  top: Card;
  discardCount: number;
  color: Color;
  turn: number;
  drawn: boolean;
  winner: number | null;
  log: string;
};

// Bentuk data yang sama akan dikirim server ke tiap pemain di Fase 2
export function viewFor(s: GameState, seat: number): PlayerView {
  return {
    seat,
    names: s.names,
    hand: s.hands[seat],
    counts: s.hands.map((h) => h.length),
    top: top(s),
    discardCount: s.discard.length,
    color: s.color,
    turn: s.turn,
    drawn: s.drawn,
    winner: s.winner,
    log: s.log,
  };
}

export function playableIndexes(v: PlayerView): boolean[] {
  const myTurn = v.turn === v.seat && v.winner === null;
  return v.hand.map(
    (card, i) => myTurn && matches(card, v.top, v.color) && (!v.drawn || i === v.hand.length - 1),
  );
}
```

```ts
// step = jarak antar tepi kiri kartu. Kurang dari lebar kartu berarti bertumpuk.
// ponytail: maksimal 2 baris; di atas ~28 kartu pada 360 px jarak bisa turun di bawah 24 px. Tambah baris ketiga kalau itu terjadi di game nyata.
export function handLayout(count: number, width: number, cardW: number, gap = 6, minStep = 24) {
  const stepFor = (n: number) => (n <= 1 ? cardW + gap : Math.min(cardW + gap, (width - cardW) / (n - 1)));
  let rows: 1 | 2 = 1;
  let perRow = count;
  if (count > 14 || stepFor(count) < minStep) {
    rows = 2;
    perRow = Math.ceil(count / 2);
  }
  return { rows, perRow, step: stepFor(perRow) };
}
```

```tsx
'use client';

import { useEffect, useRef, useState } from 'react';
import Card from './Card';
import { handLayout } from '@/lib/hand-layout';
import type { Card as CardT } from '@/lib/uno';
import s from './Hand.module.css';

type Props = { cards: CardT[]; playable: boolean[]; myTurn: boolean; onPlay: (index: number) => void };

export default function Hand({ cards, playable, myTurn, onPlay }: Props) {
  const ref = useRef<HTMLDivElement>(null);
  const [size, setSize] = useState({ width: 336, cardW: 52 });
  // pilihan terikat ke array tangan saat itu; kalau tangan berubah, pilihan otomatis hilang
  const [pick, setPick] = useState<{ cards: CardT[]; index: number } | null>(null);
  const selected = pick?.cards === cards ? pick.index : null;

  useEffect(() => {
    const el = ref.current!;
    const ro = new ResizeObserver(([entry]) => {
      // lebar kartu dibaca dari CSS supaya --card-w tetap satu-satunya sumber
      const cardW = parseFloat(getComputedStyle(el).getPropertyValue('--card-w')) || 52;
      setSize({ width: entry.contentRect.width, cardW });
    });
    ro.observe(el);
    return () => ro.disconnect();
  }, []);

  const click = (i: number) => {
    if (!playable[i]) return;
    // perangkat dengan hover (mouse): satu klik langsung main. Layar sentuh: ketuk pilih, ketuk lagi main.
    if (selected === i || matchMedia('(hover: hover)').matches) {
      setPick(null);
      onPlay(i);
    } else {
      setPick({ cards, index: i });
    }
  };

  const { perRow, step } = handLayout(cards.length, size.width, size.cardW);
  const starts: number[] = [];
  for (let i = 0; i < cards.length; i += perRow) starts.push(i);

  return (
    <div ref={ref} className={s.hand} role="group" aria-label={`Kartu kamu, ${cards.length} kartu`}>
      {starts.map((start) => (
        <div key={start} className={s.row}>
          {cards.slice(start, start + perRow).map((card, j) => {
            const i = start + j;
            return (
              <Card
                key={i}
                card={card}
                playable={playable[i]}
                dim={myTurn && !playable[i]}
                selected={selected === i}
                onClick={() => click(i)}
                style={j ? { marginLeft: step - size.cardW } : undefined}
              />
            );
          })}
        </div>
      ))}
    </div>
  );
}
```

```tsx
'use client';

import { useEffect, useReducer } from 'react';
import Table from './Table';
import { apply, botAction, newGame, viewFor, type Action, type GameState } from '@/lib/uno';
import type { User } from '@/lib/discord-user';

const BOT_DELAY = 900;
const ME = 0;

type GameAction = Action | { type: 'restart' };
const reducer = (s: GameState, a: GameAction): GameState => (a.type === 'restart' ? newGame(s.names) : apply(s, a));

// Game baru dibuat di client saja: DiscordGate baru me-render Game setelah effect selesai, jadi tidak ada beda hasil SSR.
export default function Game({ user }: { user: User }) {
  const [state, dispatch] = useReducer(reducer, [user.name, 'Bot 1', 'Bot 2', 'Bot 3'], (names) => newGame(names));

  useEffect(() => {
    if (state.winner !== null || state.turn === ME) return;
    const t = setTimeout(() => dispatch(botAction(state)), BOT_DELAY);
    return () => clearTimeout(t);
  }, [state]);

  return (
    <Table
      view={viewFor(state, ME)}
      avatarUrl={user.avatarUrl}
      onPlay={(index, color) => dispatch({ type: 'play', seat: ME, index, color })}
      onDraw={() => dispatch({ type: 'draw', seat: ME })}
      onPass={() => dispatch({ type: 'pass', seat: ME })}
      onRestart={() => dispatch({ type: 'restart' })}
    />
  );
}
```

```ts
export async function POST(req: Request) {
  const { code } = await req.json().catch(() => ({}));
  if (typeof code !== 'string' || !code) return Response.json({ error: 'code wajib diisi' }, { status: 400 });

  const res = await fetch('https://discord.com/api/oauth2/token', {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: new URLSearchParams({
      client_id: process.env.DISCORD_CLIENT_ID!,
      client_secret: process.env.DISCORD_CLIENT_SECRET!,
      grant_type: 'authorization_code',
      code,
    }).toString(),
  });
  if (!res.ok) return Response.json({ error: 'gagal menukar token' }, { status: 502 });

  const { access_token } = await res.json();
  return Response.json({ access_token });
}
```

```tsx
async function connect(): Promise<User> {
  // import dinamis supaya SDK tidak masuk bundle halaman pertama
  const { DiscordSDK } = await import('@discord/embedded-app-sdk');
  const clientId = process.env.NEXT_PUBLIC_DISCORD_CLIENT_ID!;
  sdk ??= new DiscordSDK(clientId);
  await sdk.ready();

  const { code } = await sdk.commands.authorize({
    client_id: clientId,
    response_type: 'code',
    state: '',
    prompt: 'none',
    scope: ['identify'],
  });

  // di dalam Discord semua request keluar lewat proxy /.proxy
  const res = await fetch('/.proxy/api/token', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ code }),
  });
  if (!res.ok) throw new Error(`/api/token ${res.status}`);
  const { access_token } = await res.json();

  const { user } = await sdk.commands.authenticate({ access_token });
  return displayUser(user);
}
```
